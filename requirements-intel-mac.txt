# Requirements for Macs with an Intel processor. Use with Python 3.12:
#   pip install -r requirements-intel-mac.txt
#
# PyTorch stopped publishing Intel Mac builds after 2.2.2, which supports
# Python <= 3.12 and numpy 1.x. pyannote.audio 4 needs torch >= 2.8, so this
# file uses pyannote.audio 3.3 with the pyannote/speaker-diarization-3.1 model
# (set DIARIZATION_MODEL in .env).

# Some packages have stopped publishing Intel Mac builds in their newest
# releases (e.g. sentencepiece 0.2.3). Prefer the last release that has a
# ready-made build over compiling the newest one from source.
--prefer-binary
sentencepiece<0.2.3

torch==2.2.2
torchaudio==2.2.2
numpy<2
pyannote.audio==3.3.2
# pyannote 3.x passes use_auth_token, which huggingface_hub 1.0 removed.
huggingface_hub<1.0
# Versions from late 2024 that still work with numpy 1.x. Without these pins,
# pip spends a very long time trying newer releases that need numpy 2.
lightning==2.4.0
torchmetrics==1.4.2
pyannote.core==5.0.0
pyannote.database==5.1.3
pyannote.metrics==3.2.1
